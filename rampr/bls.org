import requests
import pandas as pd
from io import StringIO

def download_qcew_state_year(state_area_code: str, year: int, 
                              annual: bool = True) -> pd.DataFrame:
    """
    Download and return a QCEW CSV slice for a given state and year.

    Parameters
    ----------
    state_area_code : str
        The 5-digit BLS area code for the state (e.g., '26000' for Michigan).
    year : int
        Four-digit year (e.g., 2024).
    annual : bool
        If True, fetch the Annual-Average data (quarter 'A'). Otherwise
        you can specify a quarterly slice (1–4).

    Returns
    -------
    pandas.DataFrame
        A DataFrame containing the requested QCEW slice.

    Raises
    ------
    HTTPError
        If the request failed.
    """
    # The BLS open data is archived by 'quarter'; for annual it's quarter 'A' ≡ 1
    quarter = "1" if annual else "1"

    # Construct the URL based on the BLS CSV slice location
    url = f"https://data.bls.gov/cew/data/api/{year}/{quarter}/area/{state_area_code}.csv"
    
    # Retrieve the CSV data
    response = requests.get(url)
    response.raise_for_status()

    # Parse the CSV text into a Pandas DataFrame
    df = pd.read_csv(StringIO(response.text))
    return df

  import requests
  import pandas as pd
  from io import StringIO

  def load_qcew_area_codes(csv_path: str) -> pd.DataFrame:
      """
      Load the full QCEW area codes CSV (from BLS Area Titles).
      The CSV must contain columns for area_code and area_title.

      Example source: 
        https://www.bls.gov/cew/classifications/areas/qcew-area-titles.htm
      """
      return pd.read_csv(csv_path, dtype=str)

  def get_county_codes_for_state(area_df: pd.DataFrame, state_fips: str) -> pd.DataFrame:
      """
      Filter area codes for counties belonging to a specific state.
      BLS county area codes have 5 total characters:
        first 2 = state FIPS, next 3 = county FIPS.
      """
      mask = area_df['area_fips'].str.startswith(state_fips) & (area_df['area_fips'].str[2:] != "000")
      return area_df.loc[mask].copy()

  def download_qcew_area(year: int, area_code: str, annual: bool = True) -> pd.DataFrame:
      """
      Download the QCEW CSV slice for a given year and specific area code.
      """
      quarter = "1" if annual else "1"
      url = f"https://data.bls.gov/cew/data/api/{year}/{quarter}/area/{area_code}.csv"
      response = requests.get(url)
      response.raise_for_status()
      df = pd.read_csv(StringIO(response.text), dtype=str)
      return df

  def download_all_counties_for_state(area_codes_csv: str, state_fips: str, year: int) -> pd.DataFrame:
      """
      Download QCEW data for all counties in a state for a given year.

      Parameters
      ----------
      area_codes_csv : str
          Path to the full QCEW area codes CSV (local file).
      state_fips : str
          2-digit state FIPS code (e.g., "06" = California).
      year : int
          Calendar year to download (e.g., 2024).

      Returns
      -------
      pandas.DataFrame
          Combined QCEW employment/wage records for all counties.
      """
      # Load the master area code list
      area_df = load_qcew_area_codes(area_codes_csv)

      # Filter for county codes in this state (exclude statewide code)
      county_df = get_county_codes_for_state(area_df, state_fips)

      combined = []
      for _, row in county_df.iterrows():
          code = row['area_fips']
          name = row['area_title']
          print(f"Downloading {name} ({code})…")
          try:
              df_county = download_qcew_area(year, code)
              df_county['county_name'] = name
              combined.append(df_county)
          except Exception as e:
              print(f"Failed {code}: {e}")

      return pd.concat(combined, ignore_index=True)

  # Example usage
  if __name__ == "__main__":
      datapath="../data/raw/employment/"
      area_codes_file = datapath+"area-titles-csv.csv"  # downloaded BLS area code CSV
      state_fips_code = "06"  # California
      year_to_fetch = 2024
      df_all_counties = download_all_counties_for_state(area_codes_file, state_fips_code, year_to_fetch)
      print(df_all_counties.head())
      fname = f"{datapath}{state_fips_code}_{year_to_fetch}.csv"
      df_all_counties.to_csv(fname)
